## 1. 项目概述与技术架构

### 1.1 项目定位

问卷系统是一个基于低代码理念的问卷设计与数据收集平台，采用前后端分离架构，支持可视化问卷编辑、多题型组件、数据收集与分析等功能。

### 1.2 技术栈全景

| 模块 | 技术栈 | 版本 | 核心职责 |
|------|--------|------|----------|
| **前端编辑器** | Vue 3 + TypeScript + Pinia + Element Plus | Vue 3.5.25 | 问卷可视化编辑、组件拖拽、实时预览 |
| **前端管理后台** | Vue 3 + TypeScript + Pinia + Arco Design | Vue 3.5.24 | 数据监控、统计分析、系统管理 |
| **后端服务** | Express + Sequelize + MySQL | Express 4.x | RESTful API、数据持久化、业务逻辑处理 |
| **数据库** | MySQL + IndexedDB | MySQL 8.x | 数据存储、本地缓存 |
| **构建工具** | Vite + pnpm | Vite 7.2.4 | 项目构建、依赖管理 |

```javascript
// SurveyController.generateSurvey 方法核心逻辑
static async generateSurvey(req, res) {
  try {
    const { surveyId, coms } = req.body;
    
    // 1. 数据验证
    if (!surveyId || !Array.isArray(coms)) {
      return res.status(400).json({ success: false, message: "请求数据无效" });
    }
    
    // 2. 数据库操作（支持事务）
    await sequelize.sync({ force: true });
    
    // 3. 问卷存储
    const survey = await Survey.create({
      survey_id: surveyId,
      title: "未命名问卷",
      description: ""
    });
    
    // 4. 组件批量存储
    const components = coms.map((com, index) => ({
      survey_id: surveyId,
      component_id: com.id,
      component_name: com.name,
      component_type: JSON.stringify(com.type),
      status: com.status,
      order_index: index
    }));
    
    await SurveyComponent.bulkCreate(components);
    
    // 5. 响应处理
    res.status(200).json({
      success: true,
      message: "在线问卷生成成功",
      data: { surveyId, componentCount: coms.length }
    });
  } catch (error) {
    // 6. 统一错误处理
    console.error("生成在线问卷失败:", error);
    res.status(500).json({
      success: false,
      message: "生成在线问卷失败",
      error: error.message
    });
  }
}
```

### 3.3 事务处理与数据一致性

#### 3.3.1 事务管理机制

```javascript
// 事务处理示例
static async createSurveyWithComponents(surveyData, components) {
  const transaction = await sequelize.transaction();
  
  try {
    // 1. 创建问卷
    const survey = await Survey.create(surveyData, { transaction });
    
    // 2. 批量创建组件
    const componentPromises = components.map(component => 
      SurveyComponent.create({
        ...component,
        survey_id: survey.survey_id
      }, { transaction })
    );
    
    await Promise.all(componentPromises);
    
    // 3. 提交事务
    await transaction.commit();
    
    return survey;
  } catch (error) {
    // 4. 回滚事务
    await transaction.rollback();
    throw error;
  }
}
```

```javascript
// multer.js 配置文件
const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    const uploadDir = 'uploads/';
    if (!fs.existsSync(uploadDir)) {
      fs.mkdirSync(uploadDir, { recursive: true });
    }
    cb(null, uploadDir);
  },
  filename: (req, file, cb) => {
    const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9);
    cb(null, file.fieldname + '-' + uniqueSuffix + path.extname(file.originalname));
  }
});

const upload = multer({
  storage: storage,
  limits: {
    fileSize: 5 * 1024 * 1024, // 5MB限制
    files: 10 // 最大文件数
  },
  fileFilter: (req, file, cb) => {
    const allowedTypes = /jpeg|jpg|png|gif/;
    const extname = allowedTypes.test(path.extname(file.originalname).toLowerCase());
    const mimetype = allowedTypes.test(file.mimetype);
    
    if (extname && mimetype) {
      return cb(null, true);
    }
    cb(new Error('只支持图片文件上传'));
  }
});
```

#### 4.3.2 分布式事务处理

```javascript
// 基于Saga模式的分布式事务
class DistributionSaga {
  constructor() {
    this.steps = [
      { name: 'validate_survey', compensation: 'rollback_validation' },
      { name: 'create_distribution_task', compensation: 'delete_task' },
      { name: 'send_via_channel', compensation: 'cancel_distribution' },
      { name: 'start_tracking', compensation: 'stop_tracking' }
    ];
  }

  async execute(surveyId, channelConfig, targets) {
    const sagaId = this.generateSagaId();
    const context = { sagaId, surveyId, channelConfig, targets };
    
    try {
      // 正向执行所有步骤
      for (const step of this.steps) {
        await this.executeStep(step.name, context);
        this.logStepCompletion(step.name, context);
      }
      
      return { success: true, sagaId };
    } catch (error) {
      // 补偿执行
      await this.compensate(context, error);
      throw error;
    }
  }

  async compensate(context, error) {
    console.error(`Saga ${context.sagaId} failed:`, error);
    
    // 逆序执行补偿操作
    for (let i = this.steps.length - 1; i >= 0; i--) {
      const step = this.steps[i];
      try {
        await this.executeStep(step.compensation, context);
      } catch (compError) {
        console.error(`Compensation ${step.compensation} failed:`, compError);
      }
    }
  }
}
```

```javascript
class RealTimeAnalytics {
  constructor(redisClient) {

      respondentId: response.respondentId,
      timestamp: Date.now(),
      status: response.status,
      metadata: JSON.stringify(response.metadata)
    });

    // 2. 更新实时计数器
    const pipeline = this.redis.pipeline();
    pipeline.hIncrBy(`survey:${response.surveyId}:stats`, 'total_responses', 1);
    pipeline.hIncrBy(`survey:${response.surveyId}:stats`, `status_${response.status}`, 1);
    
    // 3. 更新渠道统计
    if (response.channelId) {
      pipeline.hIncrBy(`channel:${response.channelId}:stats`, 'responses', 1);
    }
    
    await pipeline.exec();
  }

  async getRealTimeStats(surveyId) {
    const stats = await this.redis.hGetAll(`survey:${surveyId}:stats`);
    
    return {

      responseRate: this.calculateResponseRate(stats),
      statusDistribution: this.extractStatusDistribution(stats),
```

```typescript
class DatabaseSharding {
  private shards: number;
  
  constructor(shardCount: number = 4) {
    this.shards = shardCount;
  }
  
  getShardKey(surveyId: string): number {
    // 一致性哈希算法
    const hash = this.hashString(surveyId);
    return hash % this.shards;
  }
  
  getShardConnection(shardKey: number): Connection {
    return this.connections[shardKey];
  }
  
  async executeShardedQuery(surveyId: string, query: string, params: any[]) {
    const shardKey = this.getShardKey(surveyId);
    const connection = this.getShardConnection(shardKey);
    return connection.execute(query, params);
  }
}
```

```javascript
// 基于令牌桶的限流器
class RateLimiter {
  constructor(options = {}) {
    this.tokens = options.tokens || 100; // 令牌数量
    this.refillRate = options.refillRate || 10; // 每秒补充令牌数
    this.lastRefill = Date.now();
  }

  async acquire() {
    await this.refillTokens();
    
    if (this.tokens > 0) {
      this.tokens--;
      return true;
    }
    
    // 触发熔断
    throw new Error('Rate limit exceeded');
  }

  async refillTokens() {
    const now = Date.now();
    const elapsed = (now - this.lastRefill) / 1000;
    const tokensToAdd = Math.floor(elapsed * this.refillRate);
    
    if (tokensToAdd > 0) {
      this.tokens = Math.min(this.tokens + tokensToAdd, 100);
      this.lastRefill = now;
    }
  }
}
```
