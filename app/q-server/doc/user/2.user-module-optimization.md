## 1. 问题背景

### 1.1 模块概述

`app/q-server/src/modules/user` 模块是系统的用户认证与管理员模块，包含以下文件：

| 文件                 | 职责                               | 代码行数 |
| -------------------- | ---------------------------------- | -------- |
| `auth.service.ts`    | 登录/注册/Token/验证码核心业务逻辑 | 597      |
| `auth.routes.ts`     | 认证接口路由定义                   | 99       |
| `auth.middleware.ts` | Token 校验与权限验证中间件         | 60       |
| `admin.service.ts`   | 用户 CRUD / 系统配置管理           | 311      |
| `admin.routes.ts`    | 管理员接口路由定义                 | 113      |

### 1.2 现有安全措施

| 安全机制          | 实现状态  | 评价                    |
| ----------------- | --------- | ----------------------- |
| 密码哈希 (bcrypt) | ✅ 已实现 | cost factor = 10        |
| 防暴力破解        | ✅ 已实现 | Redis 计数 + 30分钟锁定 |
| JWT 黑名单        | ✅ 已实现 | Redis 存储 TTL          |
| 验证码一次性      | ✅ 已实现 | Redis 删除后使用        |
| 发送频率限制      | ✅ 已实现 | 1分钟最多3次            |
| 软删除            | ✅ 已实现 | `deleted_at` 字段       |
| 权限校验          | ✅ 已实现 | `verifySuperAdmin`      |
| 异步邮件发送      | ✅ 已实现 | RabbitMQ 队列           |

### 1.3 已知问题

尽管现有安全措施较为完善，经过代码审查发现以下问题需要优化：

- **安全性漏洞**：密码强度校验缺失、Refresh Token 刷新时未禁用旧 Access Token
- **可维护性问题**：类型定义未从 common 包导入、审计日志不完整
- **健壮性问题**：输入校验仅做非空检查，无格式校验、Redis 并发竞态风险
- **功能缺失**：密码重置功能未完成

## 2. 技术选型

### 2.1 为什么选择 Zod

| 特性             | Zod                               | 原生校验        | 正则表达式 |
| ---------------- | --------------------------------- | --------------- | ---------- |
| **类型推导**     | ✅ 自动从 schema 推导 TS 类型     | ❌ 需要手动定义 | ❌         |
| **运行时校验**   | ✅ 完整错误信息                   | ⚠️ 需手动实现   | ⚠️         |
| **链式 API**     | ✅ `z.string().min(8).regex(...)` | ❌              | ⚠️         |
| **错误信息**     | ✅ 自定义友好提示                 | ⚠️              | ❌         |
| **可组合性**     | ✅ schema 可嵌套复用              | ❌              | ❌         |
| **前后端复用**   | ✅ schema 可序列化                | ❌              | ❌         |
| **Fastify 集成** | ✅ 内置支持                       | ✅ 原生支持     | ❌         |

### 2.2 Zod vs 原生校验对比

#### 旧方案（原生校验）

```typescript
// auth.routes.ts
fastify.post("/login", async (request, reply) => {
  const { email, password } = request.body as { email: string; password: string };

  // ❌ 手动校验，代码冗长
  if (!email || !password) {
    return reply.sendBadRequest("邮箱和密码不能为空");
  }
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    return reply.sendBadRequest("无效的邮箱格式");
  }
  if (password.length < 8) {
    return reply.sendBadRequest("密码至少8位");
  }

  // 业务逻辑...
});
```

#### 新方案（Zod 校验）

```typescript
// schemas/user.schemas.ts
import { z } from "zod";

const emailSchema = z.string().min(1, "邮箱不能为空").email("请输入有效的邮箱地址");
const passwordSchema = z
  .string()
  .min(8, "密码至少8位")
  .regex(/[A-Z]/, "密码需包含大写字母")
  .regex(/[a-z]/, "密码需包含小写字母")
  .regex(/\d/, "密码需包含数字");

export const loginSchema = z.object({
  email: emailSchema,
  password: z.string().min(1, "密码不能为空")
});

// auth.routes.ts
fastify.post("/login", async (request, reply) => {
  // ✅ Zod 自动校验，失败返回 400
  const result = loginSchema.safeParse(request.body);
  if (!result.success) {
    return reply.sendBadRequest(result.error.errors[0].message);
  }

  const { email, password } = result.data;
  // 业务逻辑...
});
```

```typescript
// ─────────────────────────────────────────────────────────────────────────────
// 基础校验规则（可复用）
// ─────────────────────────────────────────────────────────────────────────────

/**
 * 邮箱格式校验
 * - 不能为空
 * - 符合 RFC 5322 邮箱格式
 */
export const emailSchema = z.string().min(1, "邮箱不能为空").email("请输入有效的邮箱地址");

/**
 * 用户名校验
 * - 必填
 * - 最多 50 字符
 * - 允许：中文、字母、数字、下划线、短横线
 */
export const usernameSchema = z
  .string()
  .min(1, "用户名不能为空")
  .max(50, "用户名最多50个字符")
  .regex(/^[\u4e00-\u9fa5a-zA-Z0-9_-]+$/, "用户名只能包含中文、字母、数字、下划线和短横线");

/**
 * 密码强度校验
 * - 最少 8 位

 * - 必须包含小写字母

export const passwordSchema = z
  .string()
  .min(8, "密码至少8位")
  .max(128, "密码最多128位")
  .regex(/[A-Z]/, "密码需包含大写字母")
  .regex(/[a-z]/, "密码需包含小写字母")

 * 6 位数字验证码

export const verifyCodeSchema = z
```

### 3.3 Zod 工具函数

```typescript
// src/utils/zod.ts

import { z } from "zod";
import type { FastifyReply } from "fastify";

/**
 * Zod 校验结果处理工具函数
 *
 * @param result - safeParse 返回结果
 * @param reply  - Fastify 回复对象
 * @returns 校验成功时返回解析后的数据，失败时发送 400 响应并返回 null
 */
export function parseAndRespond<T>(
  result: { success: true; data: T } | { success: false; error: z.ZodError },
  reply: FastifyReply
): T | null {
  if (!result.success) {
    const message = result.error.errors[0]?.message ?? "参数校验失败";
    return reply.status(400).send({
      data: null,
      code: 400,
      msg: message
    }) as unknown as null;
  }
  return result.data;
}

/**
 * 格式化 Zod 错误信息
 */
export function formatZodErrors(error: z.ZodError): string[] {
  return error.errors.map(e => `${e.path.join(".")}: ${e.message}`);
}

/**
 * 创建带自定义错误映射的 Zod Schema
 */
export function createEmailSchema(customError?: string) {
  return z
    .string()
    .min(1, "邮箱不能为空")
    .email(customError ?? "请输入有效的邮箱地址");
}
```

### 3.4 Fastify 路由集成

```typescript
// src/modules/user/auth.routes.ts

import type { FastifyPluginAsync } from "fastify";
import { AuthService } from "./auth.service.js";
import { authenticate } from "./auth.middleware.js";
import {
  loginSchema,
  sendCodeSchema,
  registerSchema,
  verifyRegisterSchema,
  refreshTokenSchema
} from "./schemas/user.schemas.js";
import { parseAndRespond } from "../../utils/zod.js";

const authRoutes: FastifyPluginAsync = async fastify => {
  const authService = new AuthService(fastify);

  // ── GET /status — 获取系统状态（公开） ────────────────────
  fastify.get("/status", async (_req, reply) => {
    const status = await authService.getSystemStatus();
    return reply.sendSuccess(status);
  });

  // ── POST /login — 用户登录（公开） ────────────────────────
  fastify.post("/login", async (request, reply) => {
    // ✅ Zod 自动校验请求体
    const body = parseAndRespond(loginSchema.safeParse(request.body), reply);
    if (!body) return;

    const result = await authService.login(body.email, body.password);
    return reply.sendSuccess(result, "登录成功");
  });

  // ── POST /send-code — 发送验证码（公开） ──────────────────
  fastify.post("/send-code", async (request, reply) => {
    const body = parseAndRespond(sendCodeSchema.safeParse(request.body), reply);
    if (!body) return;

    const result = await authService.sendCode(body.email, body.type);
    return reply.sendSuccess(result, "验证码已发送");
  });

  // ── POST /register — 初始化注册（公开） ────────────────────
  fastify.post("/register", async (request, reply) => {
    const body = parseAndRespond(registerSchema.safeParse(request.body), reply);
    if (!body) return;

    const result = await authService.registerAsSuperAdmin(body.email, body.password, body.username);
    return reply.sendSuccess(result, "注册成功");
  });

  // ── POST /verify-register — 验证邮箱并注册（公开） ─────────
  fastify.post("/verify-register", async (request, reply) => {
    const body = parseAndRespond(verifyRegisterSchema.safeParse(request.body), reply);
    if (!body) return;

    const result = await authService.verifyAndRegister(
      body.email,
      body.code,
      body.password,
      body.username ?? body.email.split("@")[0]
    );
    return reply.sendSuccess(result, "注册成功");
  });

  // ── POST /refresh — 刷新 Token（公开） ─────────────────────
  fastify.post("/refresh", async (request, reply) => {
    const body = parseAndRespond(refreshTokenSchema.safeParse(request.body), reply);
    if (!body) return;

    const result = await authService.refreshToken(body.refreshToken);
    return reply.sendSuccess(result, "Token 刷新成功");
  });

  // ── POST /logout — 登出（需认证） ──────────────────────────
  fastify.post("/logout", { preHandler: [authenticate] }, async (request, reply) => {
    const token = (request.headers.authorization ?? "").replace("Bearer ", "");
    await authService.logout(token);
    return reply.sendSuccess(null, "已退出登录");
  });
};

export default authRoutes;
```

## 4. 核心需求

### 4.1 安全加固

#### 4.1.1 密码强度校验（Zod 实现）

**问题描述**：
当前 `registerAsSuperAdmin()` 只验证密码非空，未校验密码复杂度。

**Zod Schema**：

```typescript
export const passwordSchema = z
  .string()
  .min(8, "密码至少8位")
  .max(128, "密码最多128位")
  .regex(/[A-Z]/, "密码需包含大写字母")
  .regex(/[a-z]/, "密码需包含小写字母")
  .regex(/\d/, "密码需包含数字");
// 可选：.regex(/[!@#$%^&*]/, "密码需包含特殊字符")
```

**验收标准**：

| 密码          | 校验结果 | 错误信息             |
| ------------- | -------- | -------------------- |
| `"Admin123!"` | ✅ 通过  | -                    |
| `"User@2024"` | ✅ 通过  | -                    |
| `"admin123"`  | ❌ 失败  | "密码需包含大写字母" |
| `"ADMIN123!"` | ❌ 失败  | "密码需包含小写字母" |
| `"AdminPass"` | ❌ 失败  | "密码需包含数字"     |
| `"Admin1"`    | ❌ 失败  | "密码至少8位"        |

#### 4.1.2 Refresh Token 刷新时禁用旧 Access Token

**需求**：刷新 Token 时，同时禁用旧的 Access Token。

**实现方案**：

```typescript
// auth.service.ts

const USER_ACCESS_PREFIX = "auth:user:access:";

async login(email: string, password: string) {
  // ... 验证逻辑 ...

  // 生成 Token
  const tokens = await this.generateTokens({ id, email, role });

  // ✅ 记录当前用户的 Access Token jti
  await this.fastify.redis.set(
    `${USER_ACCESS_PREFIX}${userId}`,
    tokens.accessJti,
    "EX",
    this.accessExpire
  );

  return { ...tokens, user };
}

async refreshToken(refreshToken: string) {
  // ...

  // ✅ 禁用旧 Access Token
  const oldJti = await this.fastify.redis.get(`${USER_ACCESS_PREFIX}${userId}`);
  if (oldJti) {
    await this.blacklistTokenByJti(oldJti, this.accessExpire);
  }

  // 生成新 Token
  const tokens = await this.generateTokens({ id, email, role });

  // ✅ 更新 jti 记录
  await this.fastify.redis.set(
    `${USER_ACCESS_PREFIX}${userId}`,
    tokens.accessJti,
    "EX",
    this.accessExpire
  );

  return { ...tokens, user };
}
```

#### 4.5.1 Redis 竞态条件修复（Lua 脚本）

```typescript
private async recordLoginFail(email: string): Promise<number> {
  const key = `${LOGIN_FAIL_PREFIX}${email}`;
  const script = `
    local count = redis.call('incr', KEYS[1])
    if count == 1 then
      redis.call('expire', KEYS[1], tonumber(ARGV[1]))
    end
    if count >= tonumber(ARGV[2]) then
      redis.call('set', KEYS[2], ARGV[3], 'ex', tonumber(ARGV[4]))
    end
    return count
  `;

  const count = await this.fastify.redis.eval(
    script, 2,
    key,
    `${LOGIN_LOCK_PREFIX}${email}`,
    LOGIN_FAIL_TTL,
    MAX_LOGIN_FAILS,
    Date.now(),
    LOGIN_LOCK_TTL
  ) as number;

  return count;
}
```

```typescript
async resetPassword(email: string, code: string, newPassword: string): Promise<void> {
  // 1. 验证验证码
  const stored = await this.fastify.redis.get(`${VERIFY_CODE_PREFIX}${email}`);
  if (!stored) {
    throw new AuthError("验证码已过期", BizCode.VERIFY_CODE_EXPIRED);
  }

  const { code: storedCode, type } = JSON.parse(stored);
  if (type !== "reset_password" || storedCode !== code) {
    throw new AuthError("验证码错误", BizCode.VERIFY_CODE_INVALID);
  }

  // 2. 删除已使用的验证码
  await this.fastify.redis.del(`${VERIFY_CODE_PREFIX}${email}`);

  // 3. 查找用户
  const user = await this.fastify.prisma.user.findFirst({
    where: { email, deleted_at: null }
  });
  if (!user) {
    throw new AuthError("用户不存在", BizCode.EMAIL_NOT_EXISTS);
  }

  // 4. 更新密码（使用 bcrypt 哈希）
  const passwordHash = await bcrypt.hash(newPassword, 10);
  await this.fastify.prisma.user.update({
    where: { id: user.id },
    data: { password_hash: passwordHash }
  });

  // 5. 记录审计日志
  await this.createAuditLog(user.id, "reset_password", "user", user.id);

  // 6. 可选：使该用户所有旧 Token 失效
  await this.fastify.redis.del(`${USER_ACCESS_PREFIX}${userId}`);
}
```

### A.2 Zod 工具函数

```typescript
// src/utils/zod.ts

import { z } from "zod";
import type { FastifyReply } from "fastify";

/**
 * 解析 Zod 校验结果并返回错误响应
 */
export function parseAndRespond<T>(
  result: { success: true; data: T } | { success: false; error: z.ZodError },
  reply: FastifyReply
): T | null {
  if (!result.success) {
    const message = result.error.errors[0]?.message ?? "参数校验失败";
    reply.status(400).send({ data: null, code: 400, msg: message });
    return null;
  }
  return result.data;
}

/**
 * 解析查询参数并返回错误响应
 */
export function parseQueryAndRespond<T>(
  result: { success: true; data: T } | { success: false; error: z.ZodError },
  reply: FastifyReply
): T | null {
  if (!result.success) {
    const messages = result.error.errors.map(e => e.path.join(".") + ": " + e.message);
    reply.status(400).send({ data: null, code: 400, msg: messages.join("; ") });
    return null;
  }
  return result.data;
}

/**
 * 格式化 Zod 错误为数组
 */
export function formatZodErrors(error: z.ZodError): Array<{ path: string; message: string }> {
  return error.errors.map(e => ({
    path: e.path.join("."),
    message: e.message
  }));
}
```
